function __kingu_shell_ready_marker --on-event fish_prompt
  builtin printf "\033]777;kingu-shell-ready\007"
  functions -e __kingu_shell_ready_marker
end
# Why captured: an unquoted (type -t codex) expands to zero words when codex is
# absent, leaving "test = file" — fish then errors instead of failing closed.
# Quoting in place is not the fix; fish never substitutes inside double quotes.
set -l __kingu_codex_type (type -t codex 2>/dev/null)
if test "$__kingu_codex_type" = file
  function codex
    if test -x "$KINGU_CODEX_LAUNCH_PREFLIGHT"
      command "$KINGU_CODEX_LAUNCH_PREFLIGHT" agent hooks prepare-codex >/dev/null 2>&1; or true
    end
    if test "$KINGU_CODEX_ISOLATE" != 0; and not string match -qr -- '^(agents|queue|--no-daemon|--remote|--remote=.*)$' $argv; and command codex --help 2>/dev/null </dev/null | string match -q -- '*--no-daemon*'
      set argv --no-daemon $argv
    end
    command codex $argv
  end
end
set -e __kingu_codex_type
set -l __kingu_claude_type (type -t claude 2>/dev/null)
if test -n "$KINGU_CLAUDE_PROFILE_POINTER"; and test "$__kingu_claude_type" = file
  function claude
    # Why: a WSL pane's pointer is relative to the guest home, which the host cannot know at spawn.
    set -l pointer (string replace -r '^~/' "$HOME/" -- "$KINGU_CLAUDE_PROFILE_POINTER")
    set -l profile (cat "$pointer" 2>/dev/null)
    if test -n "$CLAUDE_CONFIG_DIR"; and test "$CLAUDE_CONFIG_DIR" != "$KINGU_CLAUDE_INJECTED_CONFIG_DIR"
      command claude $argv
      return $status
    end
    if test -z "$profile"; and test -n "$CLAUDE_CONFIG_DIR"; and test -n "$KINGU_CLAUDE_USER_CONFIG_DIR"
      env -u KINGU_CLAUDE_INJECTED_CONFIG_DIR CLAUDE_CONFIG_DIR="$KINGU_CLAUDE_USER_CONFIG_DIR" claude $argv
      return $status
    end
    if test -z "$profile"
      env -u CLAUDE_CONFIG_DIR -u KINGU_CLAUDE_INJECTED_CONFIG_DIR claude $argv
      return $status
    end
    if not string match -qr '^(/|[A-Za-z]:)' -- "$profile"
      return 1
    end
    test -d "$profile"; or mkdir -p -m 700 -- "$profile" 2>/dev/null
    env CLAUDE_CONFIG_DIR="$profile" KINGU_CLAUDE_INJECTED_CONFIG_DIR="$profile" claude $argv
  end
end
set -e __kingu_claude_type
