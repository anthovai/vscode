# Kingu zsh shell-ready wrapper
__kingu_usable_zdotdir() {
  [[ -n "${1:-}" ]] || return 1
  # Kingu's own dir, by marker file or by the shape older builds wrote.
  [[ "$1" != */shell-ready/zsh ]] || return 1
  [[ ! -f "$1/.kingu-shell-wrapper" ]] || return 1
  # A directory holding no zsh startup file at all is not a config root,
  # whoever wrote it — and a stale value pointing at one would stop zsh from
  # ever reading the user's real .zshenv.
  local _kingu_startup_file
  for _kingu_startup_file in .zshenv .zshrc .zprofile .zlogin; do
    [[ -r "$1/$_kingu_startup_file" ]] && return 0
  done
  return 1
}
if __kingu_usable_zdotdir "${KINGU_ORIG_ZDOTDIR:-}"; then
  builtin export ZDOTDIR="$KINGU_ORIG_ZDOTDIR"
else
  builtin unset ZDOTDIR
fi
builtin unset KINGU_ORIG_ZDOTDIR KINGU_ZSHENV_SOURCE_DIR
builtin unfunction __kingu_usable_zdotdir
builtin typeset -ga _kingu_shell_features
_kingu_shell_features=(${(s:,:)${KINGU_SHELL_FEATURES:-}})
builtin unset KINGU_SHELL_FEATURES
# Why KINGU_HISTFILE is consumed HERE and not in the deferred hook: a user config
# that replaces precmd_functions wholesale drops the hook, and an exported value
# nothing will ever consume is then inherited by every child of this pane,
# including a nested Kingu (#11146). Captured non-exported, it cannot escape.
builtin typeset -g _kingu_histfile="${KINGU_HISTFILE:-}"
builtin unset KINGU_HISTFILE
__kingu_has_feature() { (( ${_kingu_shell_features[(Ie)$1]} )) }
__kingu_has_feature identity && printf "\033]777;kingu-shell-start:%s\007" "$$"
__kingu_osc133_precmd() {
  local exit_code=$?
  if [[ -n "${__kingu_in_command:-}" ]]; then
    builtin printf "\033]133;D;%s\007" "$exit_code"
    builtin unset __kingu_in_command
  fi
  builtin printf "\033]133;A\007"
}
__kingu_osc133_preexec() {
  builtin printf "\033]133;C\007"
  # Why typeset -g: a plain assignment here creates a global inside a function,
  # which prints a warning above every command under warn_create_global.
  builtin typeset -g __kingu_in_command=1
}
__kingu_deferred_line_init() {
  builtin emulate -L zsh
  (( ${+functions[__kingu_deferred_init]} )) || return 0
  local __kingu_direct_line_init=0
  [[ "${widgets[zle-line-init]:-}" == user:__kingu_deferred_line_init ]] && __kingu_direct_line_init=1
  __kingu_deferred_init
  if (( __kingu_direct_line_init && ${+widgets[zle-line-init]} )); then
    zle zle-line-init "$@"
  elif [[ "${widgets[zle-line-init]:-}" == user:__kingu_prompt_mark ]]; then
    local __kingu_prev_line_init_fn=""
    __kingu_prompt_mark "$@"
  fi
}
# Why: scheduled callbacks run after user prompt hooks without copying their function metadata.
__kingu_deferred_sched_init() {
  local __kingu_prompt_status=$?
  builtin emulate -L zsh
  (( ${+functions[__kingu_deferred_init]} )) && __kingu_deferred_init
  builtin unset __kingu_deferred_sched_armed
  builtin unfunction __kingu_deferred_sched_init
  return $__kingu_prompt_status
}
__kingu_arm_deferred_line_init() {
  builtin emulate -L zsh
  if [[ "${widgets[zle-line-init]:-}" != user:__kingu_deferred_line_init ]]; then
    if (( ${+widgets[zle-line-init]} )); then
      zle -A zle-line-init __kingu_saved_line_init
    fi
    zle -N zle-line-init __kingu_deferred_line_init
  fi
  if (( ! $+__kingu_deferred_sched_armed )) && builtin zmodload -F zsh/sched b:sched 2>/dev/null; then
    builtin sched +0 __kingu_deferred_sched_init && builtin typeset -g __kingu_deferred_sched_armed=1
  fi
}
__kingu_deferred_init() {
  # Why first: this body runs after the user's own config, so it would otherwise
  # inherit whatever options that config left set. Under NO_UNSET an unset
  # precmd_functions is fatal, and KSH_ARRAYS makes the 1-based feature lookup
  # drop whichever feature is listed first.
  builtin emulate -L zsh
  (( $+_kingu_deferred_init_done )) && return 0
  builtin typeset -g _kingu_deferred_init_done=1
  builtin typeset -g precmd_functions
  if (( ${+widgets[__kingu_saved_line_init]} )); then
    if [[ "${widgets[zle-line-init]:-}" == user:__kingu_deferred_line_init ]]; then
      zle -A __kingu_saved_line_init zle-line-init
    fi
    zle -D __kingu_saved_line_init
  elif [[ "${widgets[zle-line-init]:-}" == user:__kingu_deferred_line_init ]]; then
    zle -D zle-line-init
  fi
  if __kingu_has_feature markers; then
    precmd_functions=(${precmd_functions:/__kingu_deferred_init/__kingu_osc133_precmd})
    (( ${precmd_functions[(Ie)__kingu_osc133_precmd]} )) || precmd_functions+=(__kingu_osc133_precmd)
    preexec_functions=(__kingu_osc133_preexec ${preexec_functions[@]})
  else
    precmd_functions=(${precmd_functions:#__kingu_deferred_init})
  fi
  if [ -n "${KINGU_WSL_CLI_DIR:-}" ]; then
    if [ -x "$KINGU_WSL_CLI_DIR/${KINGU_CLI_COMMAND:-}" ]; then
      export PATH="$KINGU_WSL_CLI_DIR${PATH:+:$PATH}"
    else
      printf 'Kingu CLI unavailable: cannot run %s. Check WSL Windows-drive mount options.\n' "$KINGU_WSL_CLI_DIR/${KINGU_CLI_COMMAND:-}" >&2
    fi
  fi
  if [ -n "${KINGU_CLI_BIN_DIR:-}" ]; then
    case "${PATH:-}" in
      "$KINGU_CLI_BIN_DIR"|"$KINGU_CLI_BIN_DIR":*) ;;
      *) export PATH="$KINGU_CLI_BIN_DIR${PATH:+:$PATH}" ;;
    esac
  fi
  if __kingu_has_feature overlay; then
    # Why: ~/.zshrc can export the user's default OpenCode config after spawn.
    __kingu_restore_agent_teams_path() {
      [[ -n "${KINGU_AGENT_TEAMS_SHIM_DIR:-}" ]] || return 0
      case "$PATH" in
        "${KINGU_AGENT_TEAMS_SHIM_DIR}"|"${KINGU_AGENT_TEAMS_SHIM_DIR}:"*) return 0 ;;
      esac
      export PATH="${KINGU_AGENT_TEAMS_SHIM_DIR}:$PATH"
    }
    __kingu_restore_agent_teams_path
    [[ -n "${KINGU_OPENCODE_CONFIG_DIR:-}" ]] && export OPENCODE_CONFIG_DIR="${KINGU_OPENCODE_CONFIG_DIR}"
    if [[ -n "${KINGU_DATA_ACCOUNT_DATA_HOME:-}" ]]; then
      export XDG_DATA_HOME="${KINGU_DATA_ACCOUNT_DATA_HOME}"
      export XDG_STATE_HOME="${KINGU_DATA_ACCOUNT_STATE_HOME}"
      if [[ "${KINGU_DATA_ACCOUNT_PROVIDER:-}" == opencode ]]; then
        export OPENCODE_AUTH_CONTENT=""
        export OPENCODE_DB="opencode.db"
      fi
    fi
    [[ -n "${KINGU_MIMOCODE_HOME:-}" ]] && export MIMOCODE_HOME="${KINGU_MIMOCODE_HOME}"
    # Why: OMP does not auto-load Kingu's managed status extension; wrap only
    # interactive launch invocations so subcommands such as `omp config` keep
    # their normal argv shape.
    __kingu_omp_should_skip_extension() {
      case "${1:-}" in
        'help'|'--help'|'-h'|'--version'|'-v') return 0 ;;
        '__complete'|'acp'|'agents'|'auth-broker'|'auth-gateway'|'bench'|'commit'|'completions'|'config'|'dry-balance'|'gallery'|'grep'|'grievances'|'install'|'join'|'models'|'plugin'|'read'|'say'|'search'|'setup'|'shell'|'ssh'|'stats'|'tiny-models'|'token'|'ttsr'|'update'|'usage'|'worktree'|'q'|'wt') return 0 ;;
      esac
      return 1
    }
    __kingu_omp_cwd_is_usable() {
      local __kingu_physical_cwd
      [[ -x . ]] || return 1
      if [[ -n "${PWD:-}" && -d "${PWD:-}" ]]; then
        [[ "${PWD}" -ef . ]]
      else
        # Why compare the path: shell builtins can print a cached path for a deleted cwd.
        __kingu_physical_cwd="$(builtin pwd -P 2>/dev/null)" || return 1
        [[ -d "$__kingu_physical_cwd" && "$__kingu_physical_cwd" -ef . ]]
      fi
    }
    __kingu_omp_invoke() {
      local __kingu_use_extension="$1"
      shift
      if [[ $__kingu_use_extension -eq 1 && -n "${KINGU_OMP_STATUS_EXTENSION:-}" && -f "${KINGU_OMP_STATUS_EXTENSION}" ]]; then
        if [[ "${1:-}" == "launch" ]]; then
          shift
          command omp launch --extension "${KINGU_OMP_STATUS_EXTENSION}" "$@"
        else
          command omp --extension "${KINGU_OMP_STATUS_EXTENSION}" "$@"
        fi
      else
        command omp "$@"
      fi
    }
    __kingu_omp() {
      local __kingu_use_extension=1
      __kingu_omp_should_skip_extension "${1:-}" && __kingu_use_extension=0
      if ! __kingu_omp_cwd_is_usable; then
        local __kingu_logical_cwd="${PWD:-${KINGU_WORKTREE_PATH:-${KINGU_ROOT_PATH:-}}}"
        # Why: a restored shell can retain the deleted directory inode after its path is recreated.
        (
          if [[ -z "$__kingu_logical_cwd" ]]; then
            printf 'Kingu: OMP cannot start because no terminal working directory is available. Open a new terminal in an existing directory.\n' >&2
            return 1
          fi
          if ! builtin cd -P -- "$__kingu_logical_cwd" 2>/dev/null; then
            printf 'Kingu: OMP cannot access the terminal working directory "%s". Open a new terminal in an existing directory.\n' "$__kingu_logical_cwd" >&2
            return 1
          fi
          __kingu_omp_invoke "$__kingu_use_extension" "$@"
        )
      else
        __kingu_omp_invoke "$__kingu_use_extension" "$@"
      fi
    }
    if [[ -n "${KINGU_OMP_STATUS_EXTENSION:-}" ]]; then
      # Why the function reserved word: it suppresses alias expansion of the name, which
      # an `alias omp` otherwise rewrites at parse time, aborting the rest of the file.
      function omp { __kingu_omp "$@"; }
    fi

    # Why: Codex must keep using Kingu's runtime CODEX_HOME after rc files.
    [[ -n "${KINGU_CODEX_HOME:-}" ]] && export CODEX_HOME="${KINGU_CODEX_HOME}"
  fi
  # Why: a typed alias expands inside the shell, after pane launch prep.
  # Why unalias inside the substitution: an alias named codex makes command -v
  # report the alias text, and the subshell leaves the user's own alias intact.
  # Why || : twice — zsh alone aborts inside the substitution, but every shell's
  # assignment adopts its exit status, so an absent codex trips set -e in bash too.
  __kingu_codex_binary="$(unalias codex 2>/dev/null || :; command -v codex 2>/dev/null || :)"
  if [[ -n "${__kingu_codex_binary:-}" && -x "${__kingu_codex_binary}" ]]; then
    # Why the function reserved word: it suppresses alias expansion of the name,
    # which otherwise rewrites this header at parse time and aborts the whole file.
    function codex {
      # Why local: zsh's warn_create_global warns for each global a function creates.
      local __kingu_codex_arg __kingu_codex_isolate="${KINGU_CODEX_ISOLATE:-1}"
      if [[ -n "${KINGU_CODEX_LAUNCH_PREFLIGHT:-}" && -x "${KINGU_CODEX_LAUNCH_PREFLIGHT}" ]]; then
        "${KINGU_CODEX_LAUNCH_PREFLIGHT}" agent hooks prepare-codex >/dev/null 2>&1 || :
      fi
      for __kingu_codex_arg in "$@"; do
        case "$__kingu_codex_arg" in agents|queue|--no-daemon|--remote|--remote=*) __kingu_codex_isolate=0 ;; esac
      done
      # Why probe every launch: a cached answer goes stale across an upgrade, and 0.155 and older exit 2 on the flag.
      if [[ "$__kingu_codex_isolate" != 0 ]]; then
        case "$(command codex --help 2>/dev/null </dev/null)" in *--no-daemon*) set -- --no-daemon "$@" ;; esac
      fi
      command codex "$@"
    }
  fi
  unset __kingu_codex_binary
  __kingu_claude_binary="$(unalias claude 2>/dev/null || :; command -v claude 2>/dev/null || :)"
  if [[ -n "${KINGU_CLAUDE_PROFILE_POINTER:-}" && -n "${__kingu_claude_binary:-}" && -x "${__kingu_claude_binary}" ]]; then
    function claude {
      local __kingu_claude_home __kingu_claude_pointer="${KINGU_CLAUDE_PROFILE_POINTER:-}"
      # Why: a WSL pane's pointer is relative to the guest home, which the host cannot know at spawn.
      case "$__kingu_claude_pointer" in '~/'*) __kingu_claude_pointer="${HOME:-}/${__kingu_claude_pointer#??}" ;; esac
      __kingu_claude_home="$(cat "$__kingu_claude_pointer" 2>/dev/null || :)"
      if [ -n "${CLAUDE_CONFIG_DIR:-}" ] && [ "$CLAUDE_CONFIG_DIR" != "${KINGU_CLAUDE_INJECTED_CONFIG_DIR:-}" ]; then
        command claude "$@"; return
      fi
      # Why: Kingu's value replaced the user's own at spawn, so System default restores theirs.
      if [ -z "$__kingu_claude_home" ] && [ -n "${CLAUDE_CONFIG_DIR:-}" ] && [ -n "${KINGU_CLAUDE_USER_CONFIG_DIR:-}" ]; then
        ( unset KINGU_CLAUDE_INJECTED_CONFIG_DIR; export CLAUDE_CONFIG_DIR="$KINGU_CLAUDE_USER_CONFIG_DIR"; command claude "$@" ); return
      fi
      if [ -z "$__kingu_claude_home" ]; then
        ( unset CLAUDE_CONFIG_DIR KINGU_CLAUDE_INJECTED_CONFIG_DIR; command claude "$@" ); return
      fi
      case "$__kingu_claude_home" in /*|[A-Za-z]:*) ;; *) return 1 ;; esac
      # Why -m 700: matches the folder Kingu's setup creates; a credentials folder stays private.
      [ -d "$__kingu_claude_home" ] || mkdir -p -m 700 -- "$__kingu_claude_home" 2>/dev/null
      ( export CLAUDE_CONFIG_DIR="$__kingu_claude_home" KINGU_CLAUDE_INJECTED_CONFIG_DIR="$__kingu_claude_home"; command claude "$@" )
    }
  fi
  unset __kingu_claude_binary
  if [[ -n "${_kingu_histfile:-}" ]]; then
    HISTFILE="$_kingu_histfile"
  fi
  if __kingu_has_feature ready || __kingu_has_feature startup; then
    __kingu_emit_ready_marker=""
    __kingu_has_feature ready && __kingu_emit_ready_marker=1
    # Why: capture the prior zle-line-init so the marker chains to it. On a
    # re-source we are already the bound widget, so keep the function captured
    # the first time instead of clobbering it to empty (which would silently
    # drop the user's widget on every prompt after the second source). Only
    # user-defined widgets are chainable as plain functions; builtin/completion
    # forms (rare for zle-line-init) are left unchained.
    if [[ "${widgets[zle-line-init]:-}" == "user:__kingu_prompt_mark" ]]; then
      :
    elif (( ${+widgets[zle-line-init]} )) && [[ "${widgets[zle-line-init]}" == user:* ]]; then
      __kingu_prev_line_init_fn="${widgets[zle-line-init]#user:}"
    else
      __kingu_prev_line_init_fn=""
    fi
    __kingu_prompt_mark() {
      if [[ "${__kingu_emit_ready_marker-1}" == 1 ]]; then
        printf "\033]777;kingu-shell-ready\007"
      fi
      # Why: call the prior hook as a plain function, not an aliased widget, so
      # $WIDGET stays zle-line-init for add-zle-hook-widget dispatchers.
      if [[ -n "${__kingu_prev_line_init_fn:-}" ]]; then
        "${__kingu_prev_line_init_fn}" "$@"
      fi
      if (( ${+KINGU_POSIX_SHELL_STARTUP_COMMAND} )); then
        BUFFER="$KINGU_POSIX_SHELL_STARTUP_COMMAND"
        CURSOR=${#BUFFER}
        builtin unset KINGU_POSIX_SHELL_STARTUP_COMMAND
        zle accept-line
      fi
    }
    zle -N zle-line-init __kingu_prompt_mark
  fi
  # Why called here: we were appended during this prompt's own precmd sweep, so
  # the permanent hook has not run yet and the first prompt would lose its mark.
  __kingu_has_feature markers && __kingu_osc133_precmd
  builtin unset _kingu_shell_features _kingu_histfile
  (( $+__kingu_deferred_sched_armed )) || builtin unfunction __kingu_deferred_sched_init
  builtin unfunction __kingu_deferred_init __kingu_has_feature __kingu_arm_deferred_line_init
}
{
  builtin typeset _kingu_user_zshenv="${ZDOTDIR-$HOME}/.zshenv"
  [[ ! -r "$_kingu_user_zshenv" ]] || builtin source -- "$_kingu_user_zshenv"
} always {
  builtin unset _kingu_user_zshenv
  if (( ! $+_kingu_deferred_init_done )); then
    builtin typeset -ag precmd_functions
    (( ${precmd_functions[(Ie)__kingu_deferred_init]} )) || precmd_functions+=(__kingu_deferred_init)
    __kingu_arm_deferred_line_init
  fi
}
